function submitPayment() { let amount = document.getElementById("amount").value; let trx = document.getElementById("trx").value; let phone = localStorage.getItem("currentUser"); let transactions = JSON.parse(localStorage.getItem("transactions")) || []; transactions.push({ phone, amount, trx, date: new Date().toLocaleDateString(), status: "Pending" }); localStorage.setItem("transactions", JSON.stringify(transactions)); alert("সাবমিট হয়েছে (Pending)"); }
